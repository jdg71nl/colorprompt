cat > reset-debian-gui-black-screen-lockout.sh <<'EOF'
#!/bin/bash
# d261110 https://claude.ai/chat/a83e6504-278e-4748-a87b-2645f6649c35 "Debian RPi5 black screen lockout"
chvt 2; sleep 3; chvt 7
EOF

chmod +x reset-debian-gui-black-screen-lockout.sh

cat > /usr/local/bin/labwc-watchdog.sh <<'EOF'
#!/bin/bash
#= labwc-watchdog.sh
#= /usr/local/bin/labwc-watchdog.sh
# d261110 https://claude.ai/chat/a83e6504-278e-4748-a87b-2645f6649c35 "Debian RPi5 black screen lockout"
# Recover labwc from a stuck page flip (black screen) via a VT round-trip
export XDG_RUNTIME_DIR=/run/user/1000 WAYLAND_DISPLAY=wayland-0
pgrep -u jdg -x labwc >/dev/null || exit 0
[ "$(cat /sys/class/tty/tty0/active)" = tty7 ] || exit 0
if ! runuser -u jdg -- env XDG_RUNTIME_DIR=$XDG_RUNTIME_DIR WAYLAND_DISPLAY=$WAYLAND_DISPLAY \
     timeout 10 grim - >/dev/null 2>&1; then
  logger -t labwc-watchdog "render stall detected, doing VT round-trip"
  chvt 2; sleep 3; chvt 7
fi
EOF

sudo chmod +x /usr/local/bin/labwc-watchdog.sh

cat > /etc/systemd/system/labwc-watchdog.service <<'EOF'
#= /etc/systemd/system/labwc-watchdog.service
[Unit]
Description=labwc render-stall watchdog
[Service]
Type=oneshot
ExecStart=/usr/local/bin/labwc-watchdog.sh
EOF

cat > /etc/systemd/system/labwc-watchdog.timer <<'EOF'
#= /etc/systemd/system/labwc-watchdog.timer
[Unit]
Description=Run labwc watchdog every 5 min
[Timer]
OnBootSec=5min
OnUnitActiveSec=5min
[Install]
WantedBy=timers.target
EOF

sudo chmod +x /usr/local/bin/labwc-watchdog.sh
sudo systemctl daemon-reload && sudo systemctl enable --now labwc-watchdog.timer
